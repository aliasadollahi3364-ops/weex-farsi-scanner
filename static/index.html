/* ---------- ساخت اردربلاک از شکست ساختار ۴ ساعته ---------- */ function findOB(data, side, searchBars = 12) { if (!Array.isArray(data) || data.length < 20) return null; const pivotLen = Math.max(1, num("pivot4") || 3); const life = 200; const p = pivots(data, pivotLen, pivotLen); const zones = []; let lastBrokenHigh = -1; let lastBrokenLow = -1; for (let i = 0; i < data.length; i++) { const c = data[i]; // فقط پیوت‌هایی که در این نقطه تأیید شده‌اند const highs = p.highs.filter(x => x.index + pivotLen < i); const lows = p.lows.filter(x => x.index + pivotLen < i); const ph = highs.at(-1); const pl = lows.at(-1); // شکست صعودی ساختار if ( ph && c.close > ph.price && ph.index !== lastBrokenHigh ) { lastBrokenHigh = ph.index; const start = Math.max(0, i - Math.max(2, searchBars)); for (let j = i - 1; j >= start; j--) { if (data[j].close < data[j].open) { zones.push({ side: "BUY", top: data[j].open, bottom: data[j].low, createdAt: i, sourceIndex: j, brokenLevel: ph.price, used: false }); break; } } } // شکست نزولی ساختار if ( pl && c.close < pl.price && pl.index !== lastBrokenLow ) { lastBrokenLow = pl.index; const start = Math.max(0, i - Math.max(2, searchBars)); for (let j = i - 1; j >= start; j--) { if (data[j].close > data[j].open) { zones.push({ side: "SELL", top: data[j].high, bottom: data[j].open, createdAt: i, sourceIndex: j, brokenLevel: pl.price, used: false }); break; } } } } // حذف اردربلاک‌های منقضی یا باطل‌شده const valid = zones.filter(z => { const age = data.length - 1 - z.createdAt; if (age < 1 || age > life) return false; for (let i = z.createdAt + 1; i < data.length; i++) { if (z.side === "BUY" && data[i].close < z.bottom) { return false; } if (z.side === "SELL" && data[i].close > z.top) { return false; } } return true; }); const matching = valid.filter(z => z.side === side && !z.used); return matching.at(-1) || null; } /* ---------- تحلیل اردربلاک ۴ ساعته و ورود ۱۵ دقیقه ---------- */ async function analyze(symbol) { const limit = Math.max(100, num("limit") || 200); } const last = entryCandles.at(-1); const previous = entryCandles.at(-2); const atrValue = atr(entryCandles, num("atrLen") || 14); const p4 = pivots( htfCandles, num("pivot4") || 3, num("pivot4") || 3 ); const lastHigh = p4.highs.at(-1); const lastLow = p4.lows.at(-1); const bullishTrend = lastHigh && lastLow ? htfCandles.at(-1).close > (lastHigh.price + lastLow.price) / 2 : htfCandles.at(-1).close > htfCandles.at(-2).close; const bearishTrend = !bullishTrend; const bullOB = enabled("showBull") ? findOB(htfCandles, "BUY", num("obSearch") || 12) : null; const bearOB = enabled("showBear") ? findOB(htfCandles, "SELL", num("obSearch") || 12) : null; const entryPivots = pivots( entryCandles, num("pivot15") || 3, num("pivot15") || 3 ); const recentHigh = entryPivots.highs.at(-1); const recentLow = entryPivots.lows.at(-1); // بررسی برخورد به اردربلاک در چند کندل اخیر function findRecentTouch(ob) { if (!ob) return -1; const maxBars = Math.max(1, num("maxBarsSinceTouch") || 20); const start = Math.max(ob.createdAt, entryCandles.length - 1 - maxBars); for (let i = entryCandles.length - 1; i >= start; i--) { const c = entryCandles[i]; const bullConfirmStrength = atrValue > 0 && bullBody >= atrValue * (num("minConfirmBodyATR") || 0.20); const bearConfirmStrength = atrValue > 0 && bearBody >= atrValue * (num("minConfirmBodyATR") || 0.20); sell = sell && bearTouchIndex >= 0; } if (enabled("useFreshOB")) { buy = buy && !!bullOB && htfCandles.length - 1 - bullOB.createdAt <= (num("maxStoredOB") || 200); sell = sell && !!bearOB && htfCandles.length - 1 - bearOB.createdAt <= (num("maxStoredOB") || 200); sell = sell && last.close < last.open; } if (enabled("useConfirmStrength")) { buy = buy && bullConfirmStrength; sell = sell && bearConfirmStrength; } let signal = "WAIT"; let ob = null; if (buy) { signal = "BUY"; } const entry = last.close; const buffer = atrValue * (num("slBufferATR") || 0.15); let sl = null; let tp = null; if (signal === "BUY" && ob) { sl = enabled("useOBBoundarySL") ? ob.bottom - buffer : entry - atrValue - buffer; const risk = entry - sl; if (risk > 0) tp = entry + risk * (num("rr") || 2); } if (signal === "SELL" && ob) { sl = enabled("useOBBoundarySL") ? ob.top + buffer : entry + atrValue + buffer; const risk = sl - entry; if (risk > 0) tp = entry - risk * (num("rr") || 2); } return { symbol, signal, reason: reasons.join("، "), price: last.close, entry, sl, tp, ob, atr: atrValue, time: last.time }; }
